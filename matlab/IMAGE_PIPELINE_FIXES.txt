   In load_imagefile replace
        IMAGES{imageNum}.flicker=ones(dimY,dimX);
   by
        IMAGES{imageNum}.flicker=zeros(dimY,dimX);
   !! Search the whole file for 'flicker': refreshDeflickerCheck, hDeflickerCheck_Callback and any
   !! other place that resets IMAGES{n}.flicker to ones(...) must use zeros(...) too.
   (computeFlicker itself is unchanged: it is an additive row offset, so 0 = "no flicker".)

2) DEAD PIXELS were masked before the background subtraction and ended at -bkg. In update_image replace
            rfilt = rfilt - IMAGES{imageNum}.flicker;

            rfilt = rfilt.*IMAGES{imageNum}.deadmask;

   by
            rfilt = rfilt - IMAGES{imageNum}.flicker;
            % BACKGROUND
            rfilt = rfilt - IMAGES{imageNum}.bkg;
            % Dead pixels masking (after the background: they end at 0)
            rfilt = rfilt.*IMAGES{imageNum}.deadmask;

   In plot_image replace, in EVERY imagesc call (4 of them, plus the empty-image one if you want),
        [0 dimX]*pixelsize,[0 dimY]*pixelsize
   by
        [1 dimX]*pixelsize,[1 dimY]*pixelsize

6) GEOMETRY helpers. Replace both functions by:

    function inbounds = isROIWithin(ax, pos)
        % Tells whether the ROI is within the main axes
        % pos = [x y w h] (rectangle, ellipse) or an N x 2 array of vertices (polygon, freehand).
        % (length(pos) made a 3-vertex polygon never "within" and read a 4-vertex one as [x y w h])
        if isempty(pos); inbounds = true; return; end
        xbounds = get(ax,'XLim'); ybounds = get(ax,'YLim');
        if size(pos,2) == 2 && size(pos,1) ~= 1
            xs = pos(:,1); ys = pos(:,2);
            inbounds = all(xs >= xbounds(1)) && all(xs <= xbounds(2)) && all(ys >= ybounds(1)) && all(ys <= ybounds(2));
        else
            inbounds = pos(1) >= xbounds(1) && pos(1)+pos(3) <= xbounds(2) && pos(2) >= ybounds(1) && pos(2)+pos(4) <= ybounds(2);
        end
    end

    function inbounds = isCrossWithin(ax, pos)
        % Tells whether a crosshair element is within the main axes (limits INCLUDED: createCrosshair
        % draws the default lines exactly from XLim(1) to XLim(2) / YLim(1) to YLim(2); a strict test
        % was always false for them and the crosshair was reset at every redraw)
        % pos = [x1 y1; x2 y2] (line), [x1 y1 x2 y2] or a single point [x y].
        % (the former test indexed pos(1..4) in column order, mixing x and y for a 2x2 matrix)
        if isempty(pos); inbounds = true; return; end
        xbounds = get(ax,'XLim'); ybounds = get(ax,'YLim');
        if isequal(size(pos),[2 2])
            xs = pos(:,1); ys = pos(:,2);
        elseif numel(pos) == 2
            xs = pos(1); ys = pos(2);
        else
            xs = pos([1 3]); ys = pos([2 4]);
        end
        tolx = 1e-9*diff(xbounds); toly = 1e-9*diff(ybounds);
        inbounds = all(xs >= xbounds(1)-tolx) && all(xs <= xbounds(2)+tolx) && all(ys >= ybounds(1)-toly) && all(ys <= ybounds(2)+toly);
    end
